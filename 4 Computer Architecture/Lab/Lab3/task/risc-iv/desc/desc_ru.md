```assembly
    .data

input_addr:      .word  0x80               ; константа 0x80 по адресу 0x00 (адрес порта ввода)
output_addr:     .word  0x84               ; константа 0x84 по адресу 0x04 (адрес порта вывода)

    .text
    .org     0x200                         ; начало сегмента кода с адреса 0x200 (обход портов ввода-вывода 0x80-0x87)

_start:
    lui      sp, %hi(0x1000)               ; поместить старшие 20 бит 0x1000 в sp
    addi     sp, sp, %lo(0x1000)           ; прибавить младшие 12 бит 0x1000, sp = 0x1000 (вершина стека)

    lui      t0, %hi(input_addr)           ; взять старшие 20 бит адреса метки input_addr
    addi     t0, t0, %lo(input_addr)       ; прибавить младшие 12 бит, t0 = &input_addr (т.е. 0x00)
    lw       t0, 0(t0)                     ; t0 = *t0 = 0x80 (извлечь адрес порта ввода)
    lw       a0, 0(t0)                     ; a0 = *0x80 = n (считать входное значение, поместить в регистр параметра)

    jal      ra, count_trailing_zeros      ; ra = адрес следующей инструкции, перейти к count_trailing_zeros

    lui      t0, %hi(output_addr)          ; взять старшие 20 бит адреса метки output_addr
    addi     t0, t0, %lo(output_addr)      ; прибавить младшие 12 бит, t0 = &output_addr (т.е. 0x04)
    lw       t0, 0(t0)                     ; t0 = *t0 = 0x84 (извлечь адрес порта вывода)
    sw       a0, 0(t0)                     ; *0x84 = a0 (записать возвращаемое значение в порт вывода)

    halt                                   ; остановить выполнение

count_trailing_zeros:
    addi     sp, sp, -12                   ; sp -= 12, выделить 12 байт в стеке
    sw       ra, 0(sp)                     ; сохранить обратный адрес ra в стек [sp+0]
    sw       s0, 4(sp)                     ; сохранить счётчик s0 в стек [sp+4]
    sw       s1, 8(sp)                     ; сохранить копию n s1 в стек [sp+8]

    beqz     a0, ctz_return_32             ; если n == 0, перейти к ctz_return_32

    mv       s1, a0                        ; s1 = n (скопировать n в сохраняемый регистр)
    addi     s0, zero, 0                   ; s0 = 0 (инициализировать счётчик)

ctz_loop:
    mv       a0, s1                        ; a0 = s1 = n (подготовить 1-й параметр)
    addi     a1, zero, 0                   ; a1 = 0 (подготовить 2-й параметр: номер бита = 0)
    jal      ra, is_bit_set                ; вызвать is_bit_set(n, 0), проверить младший бит

    bnez     a0, ctz_done                  ; если возвращаемое значение a0 != 0 (младший бит = 1), выйти из цикла

    addi     s0, s0, 1                     ; count++ (увеличить счётчик на 1)
    addi     t0, zero, 1                   ; t0 = 1
    srl      s1, s1, t0                    ; s1 = s1 >> 1 (логический сдвиг n вправо на 1 бит)
    j        ctz_loop                      ; перейти к началу цикла, продолжить проверку

ctz_return_32:
    addi     s0, zero, 32                  ; s0 = 32 (при n==0 сразу вернуть 32)

ctz_done:
    mv       a0, s0                        ; a0 = s0 = count (поместить результат счёта в регистр возвращаемого значения)

    lw       s1, 8(sp)                     ; восстановить исходное значение s1 из стека [sp+8]
    lw       s0, 4(sp)                     ; восстановить исходное значение s0 из стека [sp+4]
    lw       ra, 0(sp)                     ; восстановить исходное значение ra из стека [sp+0]
    addi     sp, sp, 12                    ; sp += 12 (освободить стековый кадр)
    jr       ra                            ; перейти по адресу, на который указывает ra (вернуться в _start)

is_bit_set:
    srl      t0, a0, a1                    ; t0 = n >> bit (сдвинуть целевой бит в младший разряд)
    addi     t1, zero, 1                   ; t1 = 1
    and      a0, t0, t1                    ; a0 = t0 & 1 (взять младший бит: 0 или 1)
    jr       ra                            ; вернуться к вызывающему (ra не перезаписан, прямой переход)
```
