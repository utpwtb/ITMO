\section{Для неявно заданной функции записать многочлен Тейлора 2 порядка по
степеням $(x − x_0), (y − y_0)$}

\[
x + y + z - 1 = 2 \ln z \quad \quad M_0(2, -2, 1)
\]

\subsection{Первые производные}
\[
F(x,y,z) = x + y + z - 1 - 2 \ln z
\]
\[
F_x = 1, \quad F_y = 1, \quad F_z = 1 - \frac{2}{z}
\]

В $M_0$: $F_z = -1 \neq 0$ \\

\[
z_x = -\frac{F_x}{F_z} = 1, \quad z_y = -\frac{F_y}{F_z} = 1
\]


\subsection{Вторые производные}
\[
F_{zz} = \frac{2}{z^2}
\]
\[
z_{xx} = -\frac{\frac{2}{z^2}(z_x)^2}{F_z} = -\frac{2}{-1} = 2, \quad
z_{yy} = -\frac{\frac{2}{z^2}(z_y)^2}{F_z} = 2, \quad
z_{xy} = -\frac{\frac{2}{z^2}z_x z_y}{F_z} = 2
\]


\subsection{Многочлен Тейлора 2-го порядка}

\begin{align*}
T_2(x,y) = & f(x_0,y_0) + f_x(x_0,y_0)(x-x_0) + f_y(x_0,y_0)(y-y_0) \\
& + \frac{1}{2}f_{xx}(x_0,y_0)(x-x_0)^2 + f_{xy}(x_0,y_0)(x-x_0)(y-y_0) \\
& + \frac{1}{2}f_{yy}(x_0,y_0)(y-y_0)^2
\end{align*}

\begin{align*}
T_2(x,y) = & 1 + 1\cdot(x-2) + 1\cdot(y+2) \\
& + \frac{1}{2}\cdot 2\cdot(x-2)^2 + 2\cdot(x-2)(y+2) + \frac{1}{2}\cdot 2\cdot(y+2)^2 \\
& = x + y + 1 + (x-2)^2 + 2(x-2)(y+2) + (y+2)^2
\end{align*}

\subsection{Итоговый многочлен}

\[
\boxed{T_2(x,y) = x + y + 1 + (x-2)^2 + 2(x-2)(y+2) + (y+2)^2}
\]
